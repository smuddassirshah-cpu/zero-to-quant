- Shared generator (copy verbatim into every `prelude` that needs data; later levels copy it from here; I ran it in CPython with NumPy 2.5 and in Pyodide 0.27.7 with NumPy 2.0.2 and the outputs were identical):

```python
import numpy as np, pandas as pd

def make_bars(n=1260, seed=2301, mu=0.0003, sigma=0.012, phi=0.06):
    rng = np.random.default_rng(seed)
    eps = sigma * rng.standard_normal(n)
    r = np.empty(n)
    prev = 0.0
    for t in range(n):
        r[t] = mu + phi * prev + eps[t]        # AR(1) log returns: weak momentum
        prev = r[t] - mu
    close = 100 * np.exp(np.cumsum(r))
    prev_close = np.r_[100.0, close[:-1]]
    gap = 0.5 * r + 0.002 * rng.standard_normal(n)   # half of each day's move is overnight
    open_ = prev_close * np.exp(gap)
    w = sigma * 0.5 * np.abs(rng.standard_normal((2, n)))
    high = np.maximum(open_, close) * np.exp(w[0])
    low = np.minimum(open_, close) * np.exp(-w[1])
    volume = np.round(rng.lognormal(np.log(2e6), 0.3, n))
    idx = pd.bdate_range("2019-01-02", periods=n)
    return pd.DataFrame({"open": open_, "high": high, "low": low,
                         "close": close, "volume": volume}, index=idx)
```
