```python
def make_two_walks(seed, n=500):
    """Two independent Gaussian random walks from 0 (columns of one draw)."""
    z = np.random.default_rng(seed).standard_normal((n, 2))
    return np.cumsum(z[:, 0]), np.cumsum(z[:, 1])

def make_pair(seed, n=1000, alpha=5.0, beta=1.5, phi=0.95, sd_u=0.5, x0=40.0, vol=0.015):
    """A cointegrated pair of share prices in pounds. x starts at x0 and is a log random walk
    with daily sd vol; y = alpha + beta * x + u, u an AR(1) with coefficient phi and shock sd sd_u."""
    z = np.random.default_rng(seed).standard_normal((n, 2))
    x = x0 * np.exp(np.cumsum(vol * z[:, 0]))
    u = np.empty(n); u[0] = sd_u * z[0, 1]
    for t in range(1, n):
        u[t] = phi * u[t - 1] + sd_u * z[t, 1]
    return x, alpha + beta * x + u

def make_broken_pair(seed, n=1500, break_at=1000, sd_break=1.0, **kw):
    """make_pair, plus a random walk added to y from day break_at: the spread stops reverting."""
    x, y = make_pair(seed, n, **kw)
    z = np.random.default_rng(seed + 1000).standard_normal(n)
    return x, y + np.concatenate([np.zeros(break_at), np.cumsum(sd_break * z[break_at:])])

def make_share_pair(seed=2024, n=2520):
    """Ten years of two seeded share prices that mimic a cointegrated pair (synthetic, not real data)."""
    return make_pair(seed, n, alpha=3.0, beta=1.2, phi=0.97, sd_u=0.3, x0=25.0, vol=0.018)
```

With `make_pair` the true hedge ratio is β = 1.5 shares of x per share of y, the spread u has half-life ln 0.5 / ln 0.95 = 13.51 days, and y − 1.5x − 5 is stationary. Work in price levels (pounds) throughout so a hedge ratio is a number of shares; say once, in l5, that desks often regress log prices instead, which makes β a ratio of returns, and that the tests are the same.

```python
def rolling_fit(y, x, window):
    """alpha_t, beta_t and residual sd from OLS of y on x over days t - window + 1 to t."""
    a = np.full(len(y), np.nan); b = a.copy(); sd = a.copy()
    for t in range(window - 1, len(y)):
        xs, ys = x[t - window + 1:t + 1], y[t - window + 1:t + 1]
        bb, aa = np.polyfit(xs, ys, 1); r = ys - aa - bb * xs
        a[t], b[t], sd[t] = aa, bb, r.std(ddof=1)
    return a, b, sd

def positions(z, entry=2.0, exit=0.0, stop=4.0):
    """+1 long the spread (long y, short beta x), -1 short, 0 flat; decided at each day's close."""
    pos = np.zeros(len(z)); cur = 0
    for t in range(len(z)):
        zt = z[t]
        if np.isnan(zt): cur = 0
        elif cur == 0:
            if entry <= zt < stop: cur = -1
            elif -stop < zt <= -entry: cur = 1
        elif cur == 1 and (zt >= -exit or zt <= -stop): cur = 0
        elif cur == -1 and (zt <= exit or zt >= stop): cur = 0
        pos[t] = cur
    return pos

def backtest(y, x, window=250, entry=2.0, exit=0.0, stop=4.0, cost_bp=5.0):
    """Trade at the close of the signal day; the hedge ratio is fixed at entry; P&L in pounds
    for one share of y; cost_bp on the pounds traded in both legs at every position change."""
    a, b, sd = rolling_fit(y, x, window)
    z = (y - a - b * x) / sd
    pos = positions(z, entry, exit, stop)
    n = len(y); pnl = np.zeros(n); units = 0.0; hb = 0.0; trades = 0
    for t in range(1, n):
        pnl[t] = units * ((y[t] - y[t - 1]) - hb * (x[t] - x[t - 1]))
        if pos[t] != pos[t - 1]:
            new_hb = b[t] if pos[t] != 0 else 0.0
            traded = abs(pos[t] - units) * y[t] + abs(pos[t] * new_hb - units * hb) * x[t]
            pnl[t] -= traded * cost_bp / 1e4; trades += 1
            units, hb = pos[t], new_hb
    sharpe = pnl.mean() / pnl.std(ddof=1) * np.sqrt(252)
    return pnl, sharpe, trades
```
