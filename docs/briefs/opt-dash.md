```python
import numpy as np, pandas as pd
from math import erf, exp, log, sqrt

def norm_cdf(z):
    return 0.5 * (1 + erf(z / sqrt(2)))

def bs_price(S, K, r, sigma, T, kind="call", q=0.0):
    d1 = (log(S / K) + (r - q + sigma ** 2 / 2) * T) / (sigma * sqrt(T))
    d2 = d1 - sigma * sqrt(T)
    if kind == "call":
        return S * exp(-q * T) * norm_cdf(d1) - K * exp(-r * T) * norm_cdf(d2)
    return K * exp(-r * T) * norm_cdf(-d2) - S * exp(-q * T) * norm_cdf(-d1)

def true_iv(K, T, S=100.0, r=0.05, q=0.01):
    m = log(K / (S * exp((r - q) * T))) / sqrt(T)       # moneyness per root year
    atm = 0.22 - 0.045 * exp(-T / 0.3)                   # term structure: rises with T
    return atm - 0.08 * m + 0.10 * m * m                 # skew and smile

def make_chain(seed=2320, S=100.0, r=0.05, q=0.01):
    rng = np.random.default_rng(seed)
    rows = []
    for T in [1 / 12, 2 / 12, 3 / 12, 6 / 12, 1.0]:
        for K in range(70, 131, 5):
            for kind in ("call", "put"):
                mid = bs_price(S, K, r, true_iv(K, T, S, r, q), T, kind, q)
                half = max(0.025, 0.03 * mid)
                noise = 0.2 * half * rng.standard_normal()
                bid = max(round(mid - half + noise, 2), 0.0)
                ask = round(mid + half + noise, 2)
                rows.append([T, K, kind, bid, ask, int(rng.integers(0, 5000)), 0.0])
    df = pd.DataFrame(rows, columns=["T", "strike", "type", "bid", "ask", "volume", "age_min"])
    bad = rng.choice(len(df), 9, replace=False)
    df.loc[bad[:3], "bid"] = 0.0                                          # zero bids
    df.loc[bad[3:5], "bid"] = df.loc[bad[3:5], "ask"] + 0.05              # crossed
    df.loc[bad[5:7], "age_min"] = 240.0                                   # stale
    df.loc[bad[7:], "ask"] = (df.loc[bad[7:], "ask"] * 1.6).round(2)      # wide
    return df
```
