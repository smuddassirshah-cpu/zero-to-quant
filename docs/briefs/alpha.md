```python
import numpy as np, pandas as pd

def make_panel(n_stocks=200, n_days=1260, seed=2310):
    """Seeded stand-in for a few years of daily data on a share universe."""
    rng = np.random.default_rng(seed)
    names = [f"S{i:03d}" for i in range(n_stocks)]
    dates = pd.bdate_range("2019-01-02", periods=n_days)
    sector = rng.integers(0, 5, n_stocks)
    beta = 0.6 + 0.8 * rng.random(n_stocks)
    mkt = 0.0003 + 0.010 * rng.standard_normal(n_days)
    sec = 0.005 * rng.standard_normal((n_days, 5))
    z_mu = rng.standard_normal((n_days, n_stocks))
    z_tr = rng.standard_normal((n_days, n_stocks))
    idio = 0.015 * rng.standard_normal((n_days, n_stocks))
    mu = np.zeros(n_stocks); u = np.zeros(n_stocks)
    ret = np.empty((n_days, n_stocks))
    k = 0.0004 * np.sqrt(1 - 0.995 ** 2)
    for t in range(n_days):
        mu = 0.995 * mu + k * z_mu[t]              # slow drift: momentum
        u_new = 0.8 * u + 0.006 * z_tr[t]          # transient part: reversal
        ret[t] = beta * mkt[t] + sec[t, sector] + mu + idio[t] + (u_new - u)
        u = u_new
    # listings and delistings: 10% list late, 10% die (last day -50% to -100%)
    listed = np.ones((n_days, n_stocks), bool)
    late = rng.choice(n_stocks, n_stocks // 10, replace=False)
    for j in late:
        listed[: rng.integers(60, n_days // 2), j] = False
    dead = rng.choice(np.setdiff1d(np.arange(n_stocks), late), n_stocks // 10, replace=False)
    for j in dead:
        d = rng.integers(n_days // 4, n_days - 20)
        ret[d, j] = np.log(1 - rng.uniform(0.5, 1.0) + 1e-4)
        listed[d + 1 :, j] = False
    ret = np.where(listed, ret, np.nan)
    simple = np.expm1(ret)
    # dividends: half the shares pay 0.5% a quarter; splits: 15 two-for-one
    div_yield = np.zeros((n_days, n_stocks))
    payers = rng.random(n_stocks) < 0.5
    div_yield[63::63, :] = np.where(payers, 0.005, 0.0)
    split = np.ones((n_days, n_stocks))
    for j in rng.choice(n_stocks, 15, replace=False):
        split[rng.integers(100, n_days - 100), j] = 2.0
    price_ret = (1 + np.nan_to_num(simple)) / (1 + div_yield) - 1
    start = rng.uniform(20, 200, n_stocks)
    adj_close = start * np.cumprod(1 + price_ret, axis=0)    # split-adjusted, no dividends
    later = np.cumprod(split[::-1], axis=0)[::-1]              # splits on or after each day
    raw_close = adj_close * np.vstack([later[1:], np.ones(n_stocks)])
    # spreads and volume: bigger names are cheaper to trade
    size = rng.lognormal(0, 1, n_stocks)
    half_spread_bp = np.clip(8 / np.sqrt(size), 1, 30)
    adv_gbp = 2e7 * size
    F = lambda a: pd.DataFrame(a, index=dates, columns=names)
    return {"ret": F(simple), "raw_close": F(np.where(listed, raw_close, np.nan)),
            "adj_close": F(np.where(listed, adj_close, np.nan)),
            "dividend": F(np.where(listed, div_yield * raw_close, 0.0)),
            "split": F(split), "listed": F(listed),
            "sector": pd.Series(sector, index=names), "beta": pd.Series(beta, index=names),
            "half_spread_bp": pd.Series(half_spread_bp, index=names),
            "adv_gbp": pd.Series(adv_gbp, index=names)}
```
